cmake_minimum_required(VERSION 3.21)
project(Japan365POS VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Sql)
qt_add_executable(Japan365POS WIN32 src/main.cpp src/database.cpp src/database.h)
set_source_files_properties(sql/schema.sql PROPERTIES QT_RESOURCE_ALIAS "schema.sql")
qt_add_resources(Japan365POS "sql_resources" PREFIX "/" FILES sql/schema.sql)
target_link_libraries(Japan365POS PRIVATE Qt6::Widgets Qt6::Sql)
install(TARGETS Japan365POS RUNTIME DESTINATION .)
